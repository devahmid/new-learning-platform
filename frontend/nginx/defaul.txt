server {
  listen 443 ssl;
  server_name arrisala.fr www.arrisala.fr;

  ssl_certificate /etc/letsencrypt/live/arrisala.fr/fullchain.pem;
  ssl_certificate_key /etc/letsencrypt/live/arrisala.fr/privkey.pem;

  root /usr/share/nginx/html;
  index index.html;

  location / {
    try_files $uri $uri/ /index.html;
  }
}

server {
  listen 443 ssl;
  server_name grafana.arrisala.fr;

  ssl_certificate /etc/letsencrypt/live/arrisala.fr/fullchain.pem;
  ssl_certificate_key /etc/letsencrypt/live/arrisala.fr/privkey.pem;

  location / {
    proxy_pass http://grafana:3000;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
  }
}
server {
  listen 443 ssl;
  server_name portainer.arrisala.fr;

  ssl_certificate /etc/letsencrypt/live/arrisala.fr/fullchain.pem;
  ssl_certificate_key /etc/letsencrypt/live/arrisala.fr/privkey.pem;

  location / {
    proxy_pass http://portainer:9000;
    proxy_http_version 1.1;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;

    # Optionnel : limite le cache navigateur
    add_header Cache-Control "no-store" always;
  }
}
